Here's a minimal levelized-cost model you can run and tweak:

lcos.py·python 
def lcos(capex, cycles, rte, opex=0.02, years=15):
    """Levelized cost of storage, $/MWh delivered."""
    delivered = cycles * rte  # MWh per MWh installed
    spend = capex * 1_000 * (1 + opex * years)
    return round(spend / delivered, 1)

print("LFP   ", lcos(92, 6000, 0.88))   # 22.7
print("Na-ion", lcos(112, 5500, 0.84))  # 31.5
